Editor.js

Block-style editor with a clean, Notion-like interface. Each content type is a separate block.

Key Features

  • Block Architecture: Clean JSON with typed blocks
  • Checklist Plugin: Built-in checkbox support
  • Plugin Ecosystem: Many community plugins
  • Clean Output: Structured JSON, easy to parse
  • Modern UX: Notion-like block manipulation

Considerations

  • Block-only: No inline mixing (checkbox can't be mid-paragraph)
  • Angular: No official wrapper, manual integration needed
  • Bundle Size: ~30KB core + plugins (~10KB each)
  • Mobile: Works but not optimized for touch
  • Nesting: Limited nested structure support

Angular Integration

Manual integration required

Editor.js doesn't have an official Angular wrapper. You'll need to initialize it manually and handle zone integration.

Manual EditorJS setup with Angular
// note-editor.component.ts
import { Component, ElementRef, ViewChild, AfterViewInit, OnDestroy, output, input, NgZone } from '@angular/core';
import EditorJS from '@editorjs/editorjs';
import Checklist from '@editorjs/checklist';
import Header from '@editorjs/header';
import List from '@editorjs/list';

@Component({
  selector: 'app-note-editor',
  standalone: true,
  template: `<div #editorContainer></div>`
})
export class NoteEditorComponent implements AfterViewInit, OnDestroy {
  @ViewChild('editorContainer') editorContainer!: ElementRef;

  readonly initialData = input<EditorJSData | undefined>();
  readonly contentChange = output<EditorJSData>();

  private editor!: EditorJS;
  private ngZone = inject(NgZone);

  ngAfterViewInit() {
    // Initialize outside Angular zone to prevent change detection issues
    this.ngZone.runOutsideAngular(() => {
      this.editor = new EditorJS({
        holder: this.editorContainer.nativeElement,
        tools: {
          header: Header,
          checklist: Checklist,
          list: List,
        },
        data: this.initialData(),
        onChange: async () => {
          const data = await this.editor.save();
          this.ngZone.run(() => {
            this.contentChange.emit(data);
          });
        },
      });
    });
  }

  ngOnDestroy() {
    this.editor?.destroy();
  }

  // Extract checkboxes from editor data
  getCheckboxes(): Checkbox[] {
    return this.editor.save().then(data => {
      return data.blocks
        .filter(block => block.type === 'checklist')
        .flatMap(block => block.data.items.map((item, index) => ({
          id: `${block.id}-${index}`,
          text: item.text,
          isChecked: item.checked,
        })));
    });
  }
}

Milkdown

Plugin-driven WYSIWYG Markdown editor built on ProseMirror. Markdown-first with WYSIWYG rendering.

Key Features

  • Markdown Native: Write Markdown, see WYSIWYG
  • GFM Task Lists: Native checkbox support via - [ ] syntax
  • Plugin System: Composable, headless architecture
  • ProseMirror: Same foundation as TipTap
  • Portable: Content is standard Markdown
  • TypeScript: Full TypeScript support

Considerations

  • Angular: No official wrapper, needs manual setup
  • Bundle Size: ~50KB min+gzip (with plugins)
  • Learning Curve: Plugin composition takes time
  • Newer: Smaller community than TipTap/Quill
  • Output: Markdown string (native)

Checkbox Extraction from Markdown

Parsing GFM task list syntax
// checkbox-extractor.ts
export interface ExtractedCheckbox {
  id: string;
  text: string;
  isChecked: boolean;
  lineNumber: number;
}

export function extractCheckboxes(markdown: string): ExtractedCheckbox[] {
  const lines = markdown.split('\n');
  const checkboxes: ExtractedCheckbox[] = [];

  // GFM task list regex: - [ ] or - [x]
  const taskRegex = /^(\s*)-\s+\[([ xX])\]\s+(.+)$/;

  lines.forEach((line, index) => {
    const match = line.match(taskRegex);
    if (match) {
      checkboxes.push({
        id: `line-${index}`, // Or generate UUID
        text: match[3].trim(),
        isChecked: match[2].toLowerCase() === 'x',
        lineNumber: index,
      });
    }
  });

  return checkboxes;
}

// Usage
const markdown = `## Tasks
- [ ] First task
- [x] Completed task
- [ ] Third task`;

const checkboxes = extractCheckboxes(markdown);
// [
//   { id: 'line-1', text: 'First task', isChecked: false, lineNumber: 1 },
//   { id: 'line-2', text: 'Completed task', isChecked: true, lineNumber: 2 },
//   { id: 'line-3', text: 'Third task', isChecked: false, lineNumber: 3 },
// ]

Note: Using line numbers as IDs is fragile - if the user edits above a checkbox, the line number changes. A more robust approach would be to assign stable IDs via a custom Milkdown plugin.

Angular Integration

Manual Milkdown setup with Angular
// note-editor.component.ts
import { Component, ElementRef, ViewChild, AfterViewInit, OnDestroy, input, output, inject, NgZone } from '@angular/core';
import { Editor, rootCtx, defaultValueCtx } from '@milkdown/core';
import { commonmark } from '@milkdown/preset-commonmark';
import { gfm } from '@milkdown/preset-gfm'; // For task lists
import { listener, listenerCtx } from '@milkdown/plugin-listener';
import { slash } from '@milkdown/plugin-slash';

@Component({
  selector: 'app-note-editor',
  standalone: true,
  template: `<div #editorContainer class="milkdown-editor"></div>`
})
export class NoteEditorComponent implements AfterViewInit, OnDestroy {
  @ViewChild('editorContainer') editorContainer!: ElementRef;

  readonly initialContent = input<string>('');
  readonly contentChange = output<string>();

  private editor!: Editor;
  private ngZone = inject(NgZone);

  async ngAfterViewInit() {
    this.ngZone.runOutsideAngular(async () => {
      this.editor = await Editor.make()
        .config((ctx) => {
          ctx.set(rootCtx, this.editorContainer.nativeElement);
          ctx.set(defaultValueCtx, this.initialContent());
          ctx.get(listenerCtx).markdownUpdated((ctx, markdown) => {
            this.ngZone.run(() => {
              this.contentChange.emit(markdown);
            });
          });
        })
        .use(commonmark)
        .use(gfm) // Enables task lists
        .use(listener)
        .use(slash)
        .create();
    });
  }

  ngOnDestroy() {
    this.editor?.destroy();
  }

  // Get current markdown content
  getMarkdown(): string {
    return this.editor.action((ctx) => {
      return ctx.get(serializerCtx)(ctx.get(editorViewCtx).state.doc);
    });
  }
}

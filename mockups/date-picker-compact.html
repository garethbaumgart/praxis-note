3. Required CSS Overrides

Add to styles.css
/* ========================================
   PrimeNG DatePicker - Compact Styling
   ======================================== */

/* Reduce overall calendar size */
.p-datepicker {
  font-size: 0.8125rem;
}

/* Compact header with month/year */
.p-datepicker-header {
  padding: 0.5rem 0.25rem;
}

.p-datepicker-title {
  font-size: 0.875rem;
  font-weight: 600;
  gap: 0.25rem;
}

/* Smaller navigation buttons */
.p-datepicker-prev,
.p-datepicker-next {
  width: 1.5rem;
  height: 1.5rem;
}

.p-datepicker-prev-icon,
.p-datepicker-next-icon {
  font-size: 0.625rem;
}

/* Compact day-of-week headers */
.p-datepicker table th {
  font-size: 0.6875rem;
  font-weight: 500;
  color: var(--color-text-muted);
  padding: 0.25rem;
}

/* Smaller day cells */
.p-datepicker table td {
  padding: 0.0625rem;
}

.p-datepicker table td > span {
  width: 1.75rem;
  height: 1.75rem;
  font-size: 0.75rem;
}

/* Calendar table spacing */
.p-datepicker table {
  font-size: 0.75rem;
}

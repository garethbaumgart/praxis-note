Quill Editor

Modern WYSIWYG editor with a modular architecture. Popular choice with good documentation.

Key Features

  • Mature: Battle-tested, used by Slack, LinkedIn, Salesforce
  • Delta Format: Operational transform for real-time collab
  • Themes: Snow (toolbar) and Bubble (floating) built-in
  • API: Clean, well-documented API
  • ngx-quill: Official-ish Angular wrapper available

Considerations

  • No Task Lists: No built-in checkbox/task list support
  • Custom Formats: Need to build checkbox extension manually
  • Bundle Size: ~43KB min+gzip (full bundle)
  • Styling: Comes with opinionated styles (can override)
  • Output: Delta (JSON) or HTML

Custom Checkbox Blot Code

Implementation effort required
// checkbox-blot.ts - Custom Quill format for checkboxes
import Quill from 'quill';

const Block = Quill.import('blots/block');

class CheckboxBlot extends Block {
  static blotName = 'checkbox';
  static tagName = 'div';
  static className = 'ql-checkbox';

  static create(value: { checked: boolean; id: string }) {
    const node = super.create() as HTMLElement;
    node.setAttribute('data-checkbox-id', value.id);
    node.setAttribute('data-checked', String(value.checked));

    const checkbox = document.createElement('input');
    checkbox.type = 'checkbox';
    checkbox.checked = value.checked;
    checkbox.addEventListener('change', (e) => {
      node.setAttribute('data-checked', String((e.target as HTMLInputElement).checked));
    });

    const content = document.createElement('span');
    content.contentEditable = 'true';

    node.appendChild(checkbox);
    node.appendChild(content);
    return node;
  }

  static value(node: HTMLElement) {
    return {
      id: node.getAttribute('data-checkbox-id'),
      checked: node.getAttribute('data-checked') === 'true',
    };
  }
}

Quill.register(CheckboxBlot);

// Usage in Angular component
@Component({...})
export class NoteEditorComponent {
  @ViewChild('editor') editorRef: ElementRef;
  quill: Quill;

  ngAfterViewInit() {
    this.quill = new Quill(this.editorRef.nativeElement, {
      theme: 'snow',
      modules: {
        toolbar: [['bold', 'italic'], ['checkbox']], // custom button
      },
    });
  }

  insertCheckbox() {
    const range = this.quill.getSelection();
    this.quill.insertEmbed(range.index, 'checkbox', {
      id: crypto.randomUUID(),
      checked: false,
    });
  }
}
